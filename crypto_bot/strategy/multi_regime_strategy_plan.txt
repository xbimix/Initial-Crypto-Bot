```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Literal, Optional


RegimeType = Literal[
    "TREND",
    "RANGE",
    "BREAKOUT",
    "MOMENTUM",
    "VOLATILE",
    "LOW_VOL",
    "CHOPPY",
    "UNKNOWN",
]

SignalType = Literal["BUY", "SELL", "HOLD"]


@dataclass
class MarketState:
    price: float
    ema_fast: float          # e.g. EMA 50
    ema_slow: float          # e.g. EMA 200
    adx: float               # e.g. ADX 14
    rsi: float               # e.g. RSI 14
    atr_pct: float           # ATR / price, e.g. 0.018 = 1.8%
    volume_ratio: float      # current volume / average volume
    near_support: bool = False
    near_resistance: bool = False
    pullback_to_fast_ema: bool = False
    breakout_up: bool = False
    breakout_down: bool = False
    ema_cross_count_recent: int = 0
    breakout_failed: bool = False


@dataclass
class RegimeResult:
    regime: RegimeType
    confidence: float


@dataclass
class TradeSignal:
    regime: RegimeType
    action: SignalType
    reason: str


class RegimeEngine:
    def __init__(
        self,
        trend_adx: float = 25.0,
        range_adx: float = 20.0,
        high_vol_atr_pct: float = 0.025,
        low_vol_atr_pct: float = 0.008,
        strong_volume_ratio: float = 1.5,
        momentum_rsi_buy: float = 60.0,
        momentum_rsi_sell: float = 40.0,
        overbought_rsi: float = 70.0,
        oversold_rsi: float = 30.0,
        min_trend_gap_pct: float = 0.01,
        choppy_cross_count: int = 3,
    ) -> None:
        self.trend_adx = trend_adx
        self.range_adx = range_adx
        self.high_vol_atr_pct = high_vol_atr_pct
        self.low_vol_atr_pct = low_vol_atr_pct
        self.strong_volume_ratio = strong_volume_ratio
        self.momentum_rsi_buy = momentum_rsi_buy
        self.momentum_rsi_sell = momentum_rsi_sell
        self.overbought_rsi = overbought_rsi
        self.oversold_rsi = oversold_rsi
        self.min_trend_gap_pct = min_trend_gap_pct
        self.choppy_cross_count = choppy_cross_count

    def detect_regime(self, m: MarketState) -> RegimeResult:
        ema_gap_pct = abs(m.ema_fast - m.ema_slow) / m.price if m.price > 0 else 0.0

        # 1. Breakout first: highest urgency
        if (m.breakout_up or m.breakout_down) and m.volume_ratio >= self.strong_volume_ratio:
            confidence = min(1.0, 0.6 + (m.volume_ratio - 1.0) * 0.2)
            return RegimeResult("BREAKOUT", round(confidence, 3))

        # 2. Choppy: avoid zone
        if m.adx < self.range_adx and m.ema_cross_count_recent >= self.choppy_cross_count:
            confidence = min(1.0, 0.55 + (m.ema_cross_count_recent * 0.08))
            return RegimeResult("CHOPPY", round(confidence, 3))

        # 3. Trend
        if m.adx >= self.trend_adx and ema_gap_pct >= self.min_trend_gap_pct:
            confidence = min(1.0, 0.55 + ((m.adx - self.trend_adx) * 0.02))
            return RegimeResult("TREND", round(confidence, 3))

        # 4. High volatility
        if m.atr_pct >= self.high_vol_atr_pct:
            confidence = min(1.0, 0.5 + ((m.atr_pct - self.high_vol_atr_pct) / max(self.high_vol_atr_pct, 1e-9)))
            return RegimeResult("VOLATILE", round(confidence, 3))

        # 5. Low volatility
        if m.atr_pct <= self.low_vol_atr_pct:
            confidence = min(1.0, 0.5 + ((self.low_vol_atr_pct - m.atr_pct) / max(self.low_vol_atr_pct, 1e-9)))
            return RegimeResult("LOW_VOL", round(confidence, 3))

        # 6. Momentum
        if (m.rsi >= self.momentum_rsi_buy or m.rsi <= self.momentum_rsi_sell) and m.volume_ratio >= 1.1:
            confidence = min(1.0, 0.5 + abs(m.rsi - 50.0) / 40.0)
            return RegimeResult("MOMENTUM", round(confidence, 3))

        # 7. Range
        if m.adx < self.range_adx:
            return RegimeResult("RANGE", 0.6)

        return RegimeResult("UNKNOWN", 0.3)

    def generate_signal(self, m: MarketState, regime: Optional[RegimeType] = None) -> TradeSignal:
        detected = regime or self.detect_regime(m).regime

        if detected == "TREND":
            return self._trend_logic(m)
        if detected == "RANGE":
            return self._range_logic(m)
        if detected == "BREAKOUT":
            return self._breakout_logic(m)
        if detected == "MOMENTUM":
            return self._momentum_logic(m)
        if detected == "VOLATILE":
            return self._volatile_logic(m)
        if detected == "LOW_VOL":
            return self._low_vol_logic(m)
        if detected == "CHOPPY":
            return TradeSignal("CHOPPY", "HOLD", "Choppy market: avoid entries")
        return TradeSignal("UNKNOWN", "HOLD", "No valid regime detected")

    def _trend_logic(self, m: MarketState) -> TradeSignal:
        if m.ema_fast > m.ema_slow and m.pullback_to_fast_ema and 40 <= m.rsi <= 60:
            return TradeSignal("TREND", "BUY", "Uptrend pullback entry")
        if m.ema_fast < m.ema_slow or m.rsi >= self.overbought_rsi:
            return TradeSignal("TREND", "SELL", "Trend exit or reversal")
        return TradeSignal("TREND", "HOLD", "Trend regime but no entry trigger")

    def _range_logic(self, m: MarketState) -> TradeSignal:
        if m.near_support and m.rsi <= self.oversold_rsi:
            return TradeSignal("RANGE", "BUY", "Range buy near support")
        if m.near_resistance and m.rsi >= self.overbought_rsi:
            return TradeSignal("RANGE", "SELL", "Range sell near resistance")
        return TradeSignal("RANGE", "HOLD", "Range regime but no edge")

    def _breakout_logic(self, m: MarketState) -> TradeSignal:
        if m.breakout_up and m.volume_ratio >= self.strong_volume_ratio:
            return TradeSignal("BREAKOUT", "BUY", "Bullish breakout confirmed by volume")
        if m.breakout_down or m.breakout_failed:
            return TradeSignal("BREAKOUT", "SELL", "Bearish or failed breakout")
        return TradeSignal("BREAKOUT", "HOLD", "Breakout regime but no confirmation")

    def _momentum_logic(self, m: MarketState) -> TradeSignal:
        if m.rsi >= self.momentum_rsi_buy and m.volume_ratio >= 1.2:
            return TradeSignal("MOMENTUM", "BUY", "Bullish momentum confirmed")
        if m.rsi <= self.momentum_rsi_sell:
            return TradeSignal("MOMENTUM", "SELL", "Momentum fading or bearish")
        return TradeSignal("MOMENTUM", "HOLD", "Momentum regime but no trigger")

    def _volatile_logic(self, m: MarketState) -> TradeSignal:
        if m.breakout_up and m.volume_ratio >= self.strong_volume_ratio:
            return TradeSignal("VOLATILE", "BUY", "Volatile bullish breakout")
        if m.breakout_down or m.breakout_failed:
            return TradeSignal("VOLATILE", "SELL", "Volatile downside or failed move")
        return TradeSignal("VOLATILE", "HOLD", "High volatility without clean setup")

    def _low_vol_logic(self, m: MarketState) -> TradeSignal:
        if m.near_support:
            return TradeSignal("LOW_VOL", "BUY", "Low-volatility support buy")
        if m.near_resistance:
            return TradeSignal("LOW_VOL", "SELL", "Low-volatility resistance sell")
        return TradeSignal("LOW_VOL", "HOLD", "Low-volatility regime but no level touch")
```

```python
    "VOLATILE",

    reason: str

        momentum_rsi_sell: float = 40.0,
        overbought_rsi: float = 70.0,
        oversold_rsi: float = 30.0,
        min_trend_gap_pct: float = 0.01,
        choppy_cross_count: int = 3,
    ) -> None:

        self.range_adx = range_adx
        self.high_vol_atr_pct = high_vol_atr_pct
        self.low_vol_atr_pct = low_vol_atr_pct
        self.strong_volume_ratio = strong_volume_ratio
        self.momentum_rsi_buy = momentum_rsi_buy
        self.momentum_rsi_sell = momentum_rsi_sell
        self.overbought_rsi = overbought_rsi
        self.oversold_rsi = oversold_rsi
        self.min_trend_gap_pct = min_trend_gap_pct
        self.choppy_cross_count = choppy_cross_count

    def detect_regime(self, m: MarketState) -> RegimeResult:
        if m.price <= 0:
            return RegimeResult("UNKNOWN", 0.0, "Invalid price")

        ema_gap_pct = abs(m.ema_fast - m.ema_slow) / m.price

        # 1. Breakout
        if (m.breakout_up or m.breakout_down) and m.volume_ratio >= self.strong_volume_ratio:
            confidence = min(1.0, 0.60 + max(0.0, m.volume_ratio - 1.0) * 0.20)
            side = "upside" if m.breakout_up else "downside"
            return RegimeResult("BREAKOUT", round(confidence, 3), f"{side} breakout with volume")

        # 2. Choppy
        if m.adx < self.range_adx and m.ema_cross_count_recent >= self.choppy_cross_count:
            confidence = min(1.0, 0.55 + m.ema_cross_count_recent * 0.08)
            return RegimeResult("CHOPPY", round(confidence, 3), "Low ADX with frequent EMA crosses")

        # 3. Trend
        if m.adx >= self.trend_adx and ema_gap_pct >= self.min_trend_gap_pct:
            confidence = min(1.0, 0.55 + max(0.0, m.adx - self.trend_adx) * 0.02)
            direction = "bullish" if m.ema_fast > m.ema_slow else "bearish"
            return RegimeResult("TREND", round(confidence, 3), f"{direction} trend strength confirmed")

        # 4. Volatile
        if m.atr_pct >= self.high_vol_atr_pct:
            confidence = min(1.0, 0.50 + (m.atr_pct - self.high_vol_atr_pct) / max(self.high_vol_atr_pct, 1e-9))
            return RegimeResult("VOLATILE", round(confidence, 3), "ATR indicates elevated volatility")

        # 5. Low vol
        if m.atr_pct <= self.low_vol_atr_pct:
            confidence = min(1.0, 0.50 + (self.low_vol_atr_pct - m.atr_pct) / max(self.low_vol_atr_pct, 1e-9))
            return RegimeResult("LOW_VOL", round(confidence, 3), "ATR indicates compressed volatility")

        if active_regime == "CHOPPY":

        return TradeSignal("UNKNOWN", "HOLD", "Unknown regime: no action")

    def evaluate(self, m: MarketState) -> tuple[RegimeResult, TradeSignal]:
        regime_result = self.detect_regime(m)
        signal = self.generate_signal(m, regime_result.regime)
        return regime_result, signal
```

## 1) Replace `crypto_bot/strategy/regime.py`

```python
def _to_float(value, default=None):
    try:
        return float(value)
    except (TypeError, ValueError):
        return default


def _clamp(value: float, low: float, high: float) -> float:
    return max(low, min(high, value))


def detect_regime(snapshot: dict, regime_cfg: dict = None) -> str:
    """
    Multi-regime detector with safe fallbacks.

    Primary regimes:
    - trend_up
    - trend_down
    - range
    - breakout_up
    - breakout_down
    - momentum_up
    - momentum_down
    - volatile
    - low_vol
    - choppy
    - unknown

    Notes:
    - Works even if ADX / volume_ratio are missing.
    - Uses existing bot fields first.
    """

    regime_cfg = regime_cfg or {}

    price = _to_float(snapshot.get("price"), 0.0)
    high = _to_float(snapshot.get("high_24h"), 0.0)
    low = _to_float(snapshot.get("low_24h"), 0.0)
    atr = _to_float(snapshot.get("atr"), 0.0) or 0.0
    momentum = _to_float(snapshot.get("momentum_norm"), 0.0) or 0.0

    ema_50 = _to_float(snapshot.get("ema_50"))
    ema_200 = _to_float(snapshot.get("ema_200"))
    ema_50_slope = _to_float(snapshot.get("ema_50_slope"), 0.0) or 0.0
    ema_200_slope = _to_float(snapshot.get("ema_200_slope"), 0.0) or 0.0

    rsi = _to_float(snapshot.get("rsi"))
    adx = _to_float(snapshot.get("adx"))
    volume_ratio = _to_float(snapshot.get("volume_ratio"))
    trades = _to_float(snapshot.get("trade_count"), 0.0) or 0.0

    if price <= 0 or high is None or low is None or high <= low:
        return "unknown"

    range_width = high - low
    range_pos = _clamp((price - low) / range_width, 0.0, 1.0)
    range_pct = range_width / price
    atr_pct = atr if atr > 0 else 0.0

    # -----------------------------
    # Config
    # -----------------------------
    trend_adx = regime_cfg.get("trend_adx", 25.0)
    range_adx = regime_cfg.get("range_adx", 20.0)

    min_trend_gap_pct = regime_cfg.get("min_trend_gap_pct", 0.01)
    breakout_margin = regime_cfg.get("breakout_margin", 0.02)
    breakout_momentum = regime_cfg.get("breakout_momentum", 0.35)
    breakout_volume_ratio = regime_cfg.get("breakout_volume_ratio", 1.20)

    momentum_rsi_buy = regime_cfg.get("momentum_rsi_buy", 60.0)
    momentum_rsi_sell = regime_cfg.get("momentum_rsi_sell", 40.0)
    momentum_norm_buy = regime_cfg.get("momentum_norm_buy", 0.30)
    momentum_norm_sell = regime_cfg.get("momentum_norm_sell", -0.30)

    high_vol_atr_pct = regime_cfg.get("high_vol_atr_pct", 0.020)
    low_vol_atr_pct = regime_cfg.get("low_vol_atr_pct", 0.006)
    range_max_width_pct = regime_cfg.get("range_max_width_pct", 0.040)

    choppy_cross_threshold = regime_cfg.get("choppy_cross_threshold", 0.0025)
    choppy_momentum_abs = regime_cfg.get("choppy_momentum_abs", 0.12)

    # -----------------------------
    # Helpers
    # -----------------------------
    ema_gap_pct = 0.0
    if ema_50 is not None and ema_200 is not None and price > 0:
        ema_gap_pct = abs(ema_50 - ema_200) / price

    has_trend_alignment = (
        ema_50 is not None
        and ema_200 is not None
        and ema_gap_pct >= min_trend_gap_pct
    )

    trend_up = (
        has_trend_alignment
        and price > ema_50
        and ema_50 > ema_200
        and ema_50_slope > 0
    )
    trend_down = (
        has_trend_alignment
        and price < ema_50
        and ema_50 < ema_200
        and ema_50_slope < 0
    )

    adx_trend_ok = adx is None or adx >= trend_adx
    adx_range_ok = adx is None or adx < range_adx

    vol_breakout_ok = volume_ratio is None or volume_ratio >= breakout_volume_ratio

    breakout_up = (
        range_pos >= (1.0 - breakout_margin)
        and momentum >= breakout_momentum
        and atr_pct >= low_vol_atr_pct
        and vol_breakout_ok
    )
    breakout_down = (
        range_pos <= breakout_margin
        and momentum <= -breakout_momentum
        and atr_pct >= low_vol_atr_pct
        and vol_breakout_ok
    )

    momentum_up = (
        ((rsi is not None and rsi >= momentum_rsi_buy) or momentum >= momentum_norm_buy)
        and range_pos > 0.55
    )
    momentum_down = (
        ((rsi is not None and rsi <= momentum_rsi_sell) or momentum <= momentum_norm_sell)
        and range_pos < 0.45
    )

    # -----------------------------
    # Priority order
    # -----------------------------
    if breakout_up:
        return "breakout_up"

    if breakout_down:
        return "breakout_down"

    if trend_up and adx_trend_ok:
        return "trend_up"

    if trend_down and adx_trend_ok:
        return "trend_down"

    if atr_pct >= high_vol_atr_pct:
        return "volatile"

    if atr_pct <= low_vol_atr_pct and range_pct <= range_max_width_pct:
        return "low_vol"

    if momentum_up and not trend_down:
        return "momentum_up"

    if momentum_down and not trend_up:
        return "momentum_down"

    ema_compressed = (
        ema_50 is not None
        and ema_200 is not None
        and abs(ema_50 - ema_200) / price <= choppy_cross_threshold
    )
    if adx_range_ok and ema_compressed and abs(momentum) <= choppy_momentum_abs:
        return "choppy"

    if adx_range_ok and range_pct <= range_max_width_pct:
        return "range"

    if trades < 3 and atr_pct <= low_vol_atr_pct:
        return "low_vol"

    return "choppy"
```

## 2) Replace `crypto_bot/strategy/scoring.py`

```python
def _safe(indicators, key, default=0.0):
    value = indicators.get(key, default)
    try:
        return float(value)
    except (TypeError, ValueError):
        return default


def score_indicators(regime, indicators, range_pos):
    """
    Regime-aware score from 0..100.
    Keeps the existing interface, but stops being mean-reversion only.
    """

    rsi = _safe(indicators, "rsi", 50.0)
    momentum = _safe(indicators, "momentum", 0.0)
    structure = _safe(indicators, "structure", 0.0)
    trend_bias = _safe(indicators, "trend_bias", 0.0)
    volatility = _safe(indicators, "volatility", 0.0)
    breakout_strength = _safe(indicators, "breakout_strength", 0.0)

    score = 0.0

    if regime == "trend_up":
        if trend_bias > 0:
            score += 35
        if 40 <= rsi <= 62:
            score += 25
        elif 62 < rsi <= 70:
            score += 10
        if momentum > -0.10:
            score += 20
        if structure > 0:
            score += 20

    elif regime == "range":
        if range_pos > 0.35:
            return 0.0
        if rsi < 32:
            score += 40
        elif rsi < 42:
            score += 20
        if momentum > -0.20:
            score += 20
        if structure > 0:
            score += 20
        if range_pos <= 0.25:
            score += 20

    elif regime == "breakout_up":
        score += 35
        if breakout_strength > 0:
            score += 30
        if momentum > 0.20:
            score += 20
        if trend_bias >= 0:
            score += 15

    elif regime == "momentum_up":
        if rsi >= 58:
            score += 30
        if momentum > 0.25:
            score += 30
        if trend_bias >= 0:
            score += 20
        if structure > 0:
            score += 20

    elif regime == "volatile":
        if breakout_strength > 0:
            score += 35
        if momentum > 0.20:
            score += 20
        if volatility > 0:
            score += 15
        if trend_bias >= 0:
            score += 15
        if structure > 0:
            score += 15

    elif regime == "low_vol":
        if range_pos <= 0.25:
            score += 35
        if structure > 0:
            score += 20
        if rsi < 45:
            score += 20
        if abs(momentum) < 0.25:
            score += 15

    elif regime in ("trend_down", "breakout_down", "momentum_down"):
        score -= 40

    elif regime == "choppy":
        score -= 20

    else:
        score += 0

    return max(0.0, min(score, 100.0))
```

Your current version only feeds `rsi`, `momentum`, and `structure` into scoring. ([GitHub][3])

Inside `_compute_buy_diagnostics()`, replace the `indicators = {...}` block with this:

```python
    ema_50 = _parse_numeric(snapshot.get("ema_50"), fallback=None)
    ema_200 = _parse_numeric(snapshot.get("ema_200"), fallback=None)
    ema_50_slope = _parse_numeric(snapshot.get("ema_50_slope"), fallback=0.0) or 0.0
    volume_ratio = _parse_numeric(snapshot.get("volume_ratio"), fallback=1.0) or 1.0

    trend_bias = 0.0
    if ema_50 is not None and ema_200 is not None:
        if price > ema_50 > ema_200 and ema_50_slope > 0:
            trend_bias = 1.0
        elif price < ema_50 < ema_200 and ema_50_slope < 0:
            trend_bias = -1.0

    breakout_strength = 0.0
    if range_pos is not None:
        if range_pos >= 0.98 and momentum > 0:
            breakout_strength = 1.0
        elif range_pos <= 0.02 and momentum < 0:
            breakout_strength = -1.0

    indicators = {
        "rsi": rsi,
        "momentum": _parse_numeric(momentum, fallback=0.0),
        "structure": structure,
        "trend_bias": trend_bias,
        "volatility": volatility if volatility is not None else 0.0,
        "breakout_strength": breakout_strength * max(volume_ratio, 1.0),
    }
```
